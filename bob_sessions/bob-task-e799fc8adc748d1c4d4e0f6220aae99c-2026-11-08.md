"""Export the /api/tools OpenAPI subset to orchestrate/tools_openapi.json.

Usage:
    python scripts/export_tools_openapi.py

Env:
    PUBLIC_BASE_URL  (default: https://savealife-blr.2gwnl9m982ir.us-south.codeengine.appdomain.cloud)
"""
from __future__ import annotations

import copy
import json
import os
import sys
from pathlib import Path

# Make sure the project root is on sys.path so `app.main` can be imported.
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from app.main import app  # noqa: E402  (import after sys.path tweak)

BASE_URL = os.environ.get(
    "PUBLIC_BASE_URL",
    "https://savealife-blr.2gwnl9m982ir.us-south.codeengine.appdomain.cloud",
)

full_spec = app.openapi()
spec = copy.deepcopy(full_spec)

# Keep only paths under /api/tools
spec["paths"] = {
    path: item
    for path, item in spec.get("paths", {}).items()
    if path.startswith("/api/tools")
}

# Collect component schemas referenced by the kept paths only
# (FastAPI bundles everything; strip the 422 responses and keep the rest)
for path_item in spec["paths"].values():
    for operation in path_item.values():
        if isinstance(operation, dict):
            responses = operation.get("responses", {})
            operation["responses"] = {
                code: resp
                for code, resp in responses.items()
                if code != "422"
            }

# Metadata
spec["info"]["title"] = "SaveAlife BLR tools"
spec["info"]["version"] = "1.0.0"
spec["servers"] = [{"url": BASE_URL}]

# Write output
out_dir = Path(__file__).resolve().parents[1] / "orchestrate"
out_dir.mkdir(exist_ok=True)
out_path = out_dir / "tools_openapi.json"
out_path.write_text(json.dumps(spec, indent=2), encoding="utf-8")

# Report
operation_ids = [
    op.get("operationId")
    for path_item in spec["paths"].values()
    for op in path_item.values()
    if isinstance(op, dict) and "operationId" in op
]
print(f"Wrote {out_path}")
print("operationIds:", ", ".join(sorted(operation_ids)))

from __future__ import annotations

import pytest
from fastapi.testclient import TestClient

from app.main import app

c = TestClient(app)

SCENARIOS = ["S1", "S2", "S3", "S4", "S5"]

# Severity and need for each scenario (mirrors sim.triage applied to sc["report"] + sc["photo_read"])
SCENARIO_PARAMS = {
    "S1": ("P2", "ortho"),
    "S2": ("P1", "neuro"),
    "S3": ("P3", "general"),
    "S4": ("P1", "neuro"),
    "S5": ("P2", "ortho"),
}


# ---------------------------------------------------------------- /incident


@pytest.mark.parametrize("sid", SCENARIOS)
def test_incident_200(sid):
    r = c.get(f"/api/tools/{sid}/incident")
    assert r.status_code == 200
    data = r.json()
    for key in ("scenario_id", "location", "time", "lat", "lng", "report", "photo_read", "severity_windows_min"):
        assert key in data, f"missing key {key!r} in {sid} incident response"
    assert data["scenario_id"] == sid
    report = data["report"]
    for rk in ("text", "count", "conscious", "breathing", "bleeding"):
        assert rk in report, f"missing report key {rk!r}"
    assert isinstance(data["severity_windows_min"], int)


# ---------------------------------------------------------------- /cameras


@pytest.mark.parametrize("sid", SCENARIOS)
def test_cameras_200(sid):
    r = c.get(f"/api/tools/{sid}/cameras")
    assert r.status_code == 200
    cams = r.json()
    assert isinstance(cams, list)
    assert len(cams) > 0
    for cam in cams:
        for key in ("id", "name", "lat", "lng", "count", "state", "note"):
            assert key in cam, f"missing key {key!r} in camera entry"
        assert cam["state"] in ("LOW", "HIGH", "BLOCKED"), f"unexpected state {cam['state']!r}"


# ---------------------------------------------------------------- /routes


@pytest.mark.parametrize("sid", SCENARIOS)
def test_routes_200(sid):
    sev, need = SCENARIO_PARAMS[sid]
    r = c.get(f"/api/tools/{sid}/routes", params={"severity": sev, "need": need})
    assert r.status_code == 200
    routes = r.json()
    assert isinstance(routes, list)
    assert len(routes) > 0

            "eta_min", "blocked_at", "spec_ok", "amb_ok", "reason", "passes", "window_min",
        ):
            assert key in rt, f"missing route key {key!r} in {sid}"


def test_s2_route_a_blocked_no_passes():

    assert r.status_code == 200

    assert route_a is not None, "route A not found in S2 response"

    assert route_a["reason"].startswith("blocked"), (
